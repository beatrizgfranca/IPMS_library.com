# SISTEMA DE BIBLIOTECA VIRTUAL - BACKEND PYTHON COM SQLITE3
# ==============================================================================
import sqlite3
from datetime import datetime, timedelta

def inicializar_banco():
    conn = sqlite3.connect('biblioteca_virtual.db')
    cursor = conn.cursor()
    
    # Tabela de Livros (Gêneros Tradicionais)
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS livros (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            titulo TEXT NOT NULL,
            autor TEXT NOT NULL,
            categoria TEXT NOT NULL,
            isbn TEXT UNIQUE NOT NULL,
            quantidade INTEGER NOT NULL,
            descricao TEXT
        )
    ''')
    
    # Tabela de Usuários
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS usuarios (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            nome TEXT NOT NULL,
            email TEXT UNIQUE NOT NULL,
            tipo TEXT NOT NULL
        )
    ''')
    
    # Tabela de Empréstimos
    cursor.execute('''
        CREATE TABLE IF NOT EXISTS emprestimos (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            usuario_id INTEGER,
            livro_id INTEGER,
            data_emprestimo TEXT,
            data_devolucao_prevista TEXT,
            devolvido INTEGER DEFAULT 0,
            FOREIGN KEY (usuario_id) REFERENCES usuarios (id),
            FOREIGN KEY (livro_id) REFERENCES livros (id)
        )
    ''')
    
    conn.commit()
    conn.close()

# Funções CRUD de Banco de Dados
def cadastrar_livro(titulo, autor, categoria, isbn, quantidade, descricao=""):
    conn = sqlite3.connect('biblioteca_virtual.db')
    cursor = conn.cursor()
    try:
        cursor.execute('''
            INSERT INTO livros (titulo, autor, categoria, isbn, quantidade, descricao)
            VALUES (?, ?, ?, ?, ?, ?)
        ''', (titulo, autor, categoria, isbn, quantidade, descricao))
        conn.commit()
        print(f"[Sucesso] Livro '{titulo}' cadastrado com sucesso!")
    except sqlite3.IntegrityError:
        print("[Erro] ISBN já existente.")
    finally:
        conn.close()

def pesquisar_livros(termo="", categoria="ALL"):
    conn = sqlite3.connect('biblioteca_virtual.db')
    cursor = conn.cursor()
    
    query = "SELECT * FROM livros WHERE (titulo LIKE ? OR autor LIKE ? OR categoria LIKE ?)"
    params = [f"%{termo}%", f"%{termo}%", f"%{termo}%"]
    
    if categoria != "ALL":
        query += " AND categoria = ?"
        params.append(categoria)
        
    cursor.execute(query, params)
    livros = cursor.fetchall()
    conn.close()
    return livros

def emprestar_livro(usuario_id, livro_id, dias=14):
    conn = sqlite3.connect('biblioteca_virtual.db')
    cursor = conn.cursor()
    
    # Verificar quantidade disponível
    cursor.execute("SELECT quantidade FROM livros WHERE id = ?", (livro_id,))
    livro = cursor.fetchone()
    
    if livro and livro[0] > 0:
        data_hoje = datetime.now().strftime("%Y-%m-%d")
        data_devolucao = (datetime.now() + timedelta(days=dias)).strftime("%Y-%m-%d")
        
        # Registrar empréstimo e decrementar estoque
        cursor.execute('''
            INSERT INTO emprestimos (usuario_id, livro_id, data_emprestimo, data_devolucao_prevista)
            VALUES (?, ?, ?, ?)
        ''', (usuario_id, livro_id, data_hoje, data_devolucao))
        
        cursor.execute("UPDATE livros SET quantidade = quantidade - 1 WHERE id = ?", (livro_id,))
        conn.commit()
        print("[Sucesso] Empréstimo registrado com sucesso!")
    else:
        print("[Erro] Livro indisponível no estoque.")
    conn.close()

# Execução Demonstrativa
if __name__ == "__main__":
    inicializar_banco()
    cadastrar_livro("Dom Casmurro", "Machado de Assis", "Romance", "9788535902777", 4, "Clássico da literatura brasileira.")
    cadastrar_livro("Python Fluente", "Luciano Ramalho", "Tecnologia", "9788575225127", 3, "Programação com a linguagem Python.")
    print("Banco de dados inicializado e atualizado!")
